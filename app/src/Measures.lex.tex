\begin{lexlean}{Measures}
\useglossary{lexlean.std.bool@1.2.0}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{prism.arch@1.0.0}
\importmodule{Catalog}
\importmodule{Foundation.DevRel.V1.Model}
\importmodule{Foundation.Experience.V1.Model}
\importmodule{Users}
\title{Measurement methods of the model}
\begin{semanticmodule}
definition{
  name: emptyRows,
  parameters: [rows : (List Foundation.DevRel.V1.Model::Covers)],
  result: Nat,
  body: (match rows
    | List.nil => 0
    | List.cons row rest => match{
      branches: [
        {body: (1 + (emptyRows rest)), binders: [], constructor: List.nil},
        {body: (emptyRows rest), binders: [first, others], constructor: List.cons}
      ],
      scrutinee: (project row resources)
    }),
  recursive_argument: rows
}

definition{
  name: undeclaredIn,
  parameters: [resourceCount : Nat, values : (List Nat)],
  result: Nat,
  body: (match values
    | List.nil => 0
    | List.cons value rest => ((prim subtract 1 (prim subtract resourceCount value : Nat) : Nat) + (undeclaredIn resourceCount rest))),
  recursive_argument: values
}

definition{
  name: undeclaredRows,
  parameters: [resourceCount : Nat, rows : (List Foundation.DevRel.V1.Model::Covers)],
  result: Nat,
  body: (match rows
    | List.nil => 0
    | List.cons row rest => ((undeclaredIn resourceCount (project row resources)) + (undeclaredRows resourceCount rest))),
  recursive_argument: rows
}

definition{
  name: uncoveredRecords,
  parameters: [],
  result: Nat,
  body: ((emptyRows (Catalog::covers)) + (prim subtract (Foundation.DevRel.V1.Model::recordTotal (Catalog::records)) (Foundation.DevRel.V1.Model::rowCount (Catalog::covers)) : Nat))
}

definition{
  name: undeclaredReferences,
  parameters: [],
  result: Nat,
  body: (undeclaredRows (Foundation.DevRel.V1.Model::resourceTotal (Catalog::resources)) (Catalog::covers))
}

definition{
  name: same,
  parameters: [a : Nat, b : Nat],
  result: Nat,
  body: (prim subtract 1 ((prim subtract a b : Nat) + (prim subtract b a : Nat)) : Nat)
}

definition{
  name: smaller,
  parameters: [a : Nat, b : Nat],
  result: Nat,
  body: (prim subtract a (prim subtract a b : Nat) : Nat)
}

definition{
  name: goalsOf,
  parameters: [user : Nat, goals : (List Foundation.Experience.V1.Model::Goal)],
  result: Nat,
  body: (match goals
    | List.nil => 0
    | List.cons goal rest => ((same (project goal user) user) + (goalsOf user rest))),
  recursive_argument: goals
}

definition{
  name: fewestGoals,
  parameters: [
    user : Nat,
    groups : (List Foundation.Experience.V1.Model::UserGroup),
    goals : (List Foundation.Experience.V1.Model::Goal)
  ],
  result: Nat,
  body: (match groups
    | List.nil => 0
    | List.cons group rest => (match rest
      | List.nil => (goalsOf user goals)
      | List.cons next others => (smaller (goalsOf user goals) (fewestGoals (user + 1) rest goals)))),
  recursive_argument: groups
}

definition{
  name: journeysPerStakeholder,
  parameters: [],
  result: Nat,
  body: (fewestGoals 0 (Users::groups) (Users::goals))
}
\end{semanticmodule}
\end{lexlean}
