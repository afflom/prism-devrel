definition{
  name: appendRecords,
  parameters: [front : (List SubjectRecord), back : (List SubjectRecord)],
  result: (List SubjectRecord),
  body: (match front
    | List.nil => back
    | List.cons item rest => (cons item (appendRecords rest back))),
  recursive_argument: front
}

definition{
  name: appendRows,
  parameters: [front : (List Covers), back : (List Covers)],
  result: (List Covers),
  body: (match front | List.nil => back | List.cons row rest => (cons row (appendRows rest back))),
  recursive_argument: front
}

definition{
  name: allBelow,
  parameters: [bound : Nat, values : (List Nat)],
  result: Bool,
  body: (match values
    | List.nil => true
    | List.cons value rest => ((value <? bound) && (allBelow bound rest))),
  recursive_argument: values
}

definition{
  name: AllBelow,
  parameters: [bound : Nat, values : (List Nat)],
  result: Prop,
  body: (match values
    | List.nil => (bound = bound)
    | List.cons value rest => ((value < bound) /\ (AllBelow bound rest))),
  recursive_argument: values
}

theorem{
  name: allBelow_sound_complete,
  parameters: [bound : Nat, values : (List Nat)],
  statement: (((allBelow bound values) = true) <-> (AllBelow bound values)),
  proof: boolean_reflection{
    reflection: list{
      boolean_definition: allBelow,
      comparison: nat_blt,
      parameter: bound,
      proposition_definition: AllBelow,
      values: values
    }
  }
}

definition{
  name: nonEmpty,
  parameters: [values : (List Nat)],
  result: Bool,
  body: (match values | List.nil => false | List.cons value rest => true)
}

definition{
  name: rowsCovered,
  parameters: [position : Nat, rows : (List Covers)],
  result: Bool,
  body: (match rows
    | List.nil => true
    | List.cons row rest => (((project row record) =? position) && ((nonEmpty (project row resources)) && (rowsCovered (position + 1) rest)))),
  recursive_argument: rows
}

definition{
  name: rowCount,
  parameters: [rows : (List Covers)],
  result: Nat,
  body: (match rows | List.nil => 0 | List.cons row rest => (1 + (rowCount rest))),
  recursive_argument: rows
}

definition{
  name: rowsDeclared,
  parameters: [resourceCount : Nat, rows : (List Covers)],
  result: Bool,
  body: (match rows
    | List.nil => true
    | List.cons row rest => ((allBelow resourceCount (project row resources)) && (rowsDeclared resourceCount rest))),
  recursive_argument: rows
}

definition{
  name: resourcesNumbered,
  parameters: [position : Nat, resources : (List Resource)],
  result: Bool,
  body: (match resources
    | List.nil => true
    | List.cons resource rest => (((project resource index) =? position) && (resourcesNumbered (position + 1) rest))),
  recursive_argument: resources
}

definition{
  name: recordsNumbered,
  parameters: [position : Nat, records : (List SubjectRecord)],
  result: Bool,
  body: (match records
    | List.nil => true
    | List.cons item rest => (((project item index) =? position) && (recordsNumbered (position + 1) rest))),
  recursive_argument: records
}

definition{
  name: resourceTotal,
  parameters: [resources : (List Resource)],
  result: Nat,
  body: (match resources | List.nil => 0 | List.cons resource rest => (1 + (resourceTotal rest))),
  recursive_argument: resources
}

definition{
  name: recordTotal,
  parameters: [records : (List SubjectRecord)],
  result: Nat,
  body: (match records | List.nil => 0 | List.cons item rest => (1 + (recordTotal rest))),
  recursive_argument: records
}

definition{
  name: everyRecordCovered,
  parameters: [subject : Subject, records : (List SubjectRecord), rows : (List Covers)],
  result: Bool,
  body: (((recordTotal records) =? (project subject recordCount)) && ((recordsNumbered 0 records) && (((rowCount rows) =? (project subject recordCount)) && (rowsCovered 0 rows))))
}

definition{
  name: coversOnlyDeclared,
  parameters: [resources : (List Resource), rows : (List Covers)],
  result: Bool,
  body: ((resourcesNumbered 0 resources) && (rowsDeclared (resourceTotal resources) rows))
}

definition{
  name: catalogValid,
  parameters: [
    subject : Subject,
    records : (List SubjectRecord),
    resources : (List Resource),
    rows : (List Covers)
  ],
  result: Bool,
  body: ((everyRecordCovered subject records rows) && (coversOnlyDeclared resources rows))
}

definition{
  name: withoutFirstCover,
  parameters: [rows : (List Covers)],
  result: (List Covers),
  body: (match rows
    | List.nil => (nil Covers)
    | List.cons row rest => (cons (record Covers {record := (project row record), resources := (nil Nat)}) rest))
}

definition{
  name: withUndeclaredCover,
  parameters: [resourceCount : Nat, rows : (List Covers)],
  result: (List Covers),
  body: (match rows
    | List.nil => (nil Covers)
    | List.cons row rest => (cons (record Covers {record := (project row record), resources := (cons resourceCount (project row resources))}) rest))
}

definition{
  name: memberNat,
  parameters: [wanted : Nat, values : (List Nat)],
  result: Bool,
  body: (match values
    | List.nil => false
    | List.cons value rest => ((value =? wanted) || (memberNat wanted rest))),
  recursive_argument: values
}

definition{
  name: teaches,
  parameters: [resource : Nat, rows : (List Covers)],
  result: Bool,
  body: (match rows
    | List.nil => false
    | List.cons row rest => ((memberNat resource (project row resources)) || (teaches resource rest))),
  recursive_argument: rows
}

definition{
  name: recordAt,
  parameters: [position : Nat, records : (List SubjectRecord)],
  result: (Option SubjectRecord),
  body: (match records
    | List.nil => (ctor Option.none [SubjectRecord])
    | List.cons item rest => (if ((project item index) =? position) then (ctor Option.some [SubjectRecord] item) else (recordAt position rest))),
  recursive_argument: records
}

definition{
  name: rowAt,
  parameters: [position : Nat, rows : (List Covers)],
  result: (List Nat),
  body: (match rows
    | List.nil => (nil Nat)
    | List.cons row rest => (if ((project row record) =? position) then (project row resources) else (rowAt position rest))),
  recursive_argument: rows
}

definition{
  name: facetName,
  parameters: [facet : Facet],
  result: String,
  body: (match facet
    | Facet.architecture => "architecture"
    | Facet.design => "design"
    | Facet.features => "features"
    | Facet.security => "security"
    | Facet.quality => "quality"
    | Facet.behavior => "behavior"
    | Facet.system => "system"
    | Facet.extension => "extension"
    | Facet.application => "application")
}

definition{
  name: slugAt,
  parameters: [position : Nat, resources : (List Resource)],
  result: String,
  body: (match resources
    | List.nil => ""
    | List.cons resource rest => (if ((project resource index) =? position) then (project resource slug) else (slugAt position rest))),
  recursive_argument: resources
}

definition{
  name: pageLines,
  parameters: [positions : (List Nat), resources : (List Resource)],
  result: String,
  body: (match positions
    | List.nil => ""
    | List.cons position rest => (prim join (cons "\npage " (cons (slugAt position resources) (cons (pageLines rest resources) (nil String)))) "" : String)),
  axioms: [propext],
  recursive_argument: positions
}

definition{
  name: answerFor,
  parameters: [
    position : Nat,
    records : (List SubjectRecord),
    resources : (List Resource),
    rows : (List Covers)
  ],
  result: String,
  body: (match (recordAt position records)
    | Option.none => "refused unknown-record"
    | Option.some item => (prim join (cons "record " (cons (project item id) (cons "\nfacet " (cons (facetName (project item facet)) (cons (pageLines (rowAt position rows) resources) (nil String)))))) "" : String)),
  axioms: [propext]
}

definition{
  name: digitOf,
  parameters: [octet : UInt8],
  result: (Option Nat),
  body: (if (prim equal octet (48 : UInt8) : Bool) then (ctor Option.some [Nat] 0) else (if (prim equal octet (49 : UInt8) : Bool) then (ctor Option.some [Nat] 1) else (if (prim equal octet (50 : UInt8) : Bool) then (ctor Option.some [Nat] 2) else (if (prim equal octet (51 : UInt8) : Bool) then (ctor Option.some [Nat] 3) else (if (prim equal octet (52 : UInt8) : Bool) then (ctor Option.some [Nat] 4) else (if (prim equal octet (53 : UInt8) : Bool) then (ctor Option.some [Nat] 5) else (if (prim equal octet (54 : UInt8) : Bool) then (ctor Option.some [Nat] 6) else (if (prim equal octet (55 : UInt8) : Bool) then (ctor Option.some [Nat] 7) else (if (prim equal octet (56 : UInt8) : Bool) then (ctor Option.some [Nat] 8) else (if (prim equal octet (57 : UInt8) : Bool) then (ctor Option.some [Nat] 9) else (ctor Option.none [Nat])))))))))))
}

definition{
  name: digitsValue,
  parameters: [value : Bytes, offset : Nat, remaining : Nat, accumulated : Nat],
  result: (Option Nat),
  body: (match remaining
    | Nat.zero => (ctor Option.some [Nat] accumulated)
    | Nat.succ rest => (match (prim index value offset : (Option UInt8))
      | Option.none => (ctor Option.some [Nat] accumulated)
      | Option.some octet => (match (digitOf octet)
        | Option.none => (ctor Option.none [Nat])
        | Option.some digit => (digitsValue value (offset + 1) rest ((prim multiply accumulated 10 : Nat) + digit))))),
  recursive_argument: remaining
}

definition{
  name: requestIndex,
  parameters: [value : Bytes, digits : Nat],
  result: (Option Nat),
  body: (if (((prim length value : Nat) =? 0) || (digits <? (prim length value : Nat))) then (ctor Option.none [Nat]) else (digitsValue value 0 (prim length value : Nat) 0))
}

definition{
  name: lookup,
  parameters: [
    value : Bytes,
    digits : Nat,
    records : (List SubjectRecord),
    resources : (List Resource),
    rows : (List Covers)
  ],
  result: String,
  body: (match (requestIndex value digits)
    | Option.none => "error invalid-request"
    | Option.some position => (answerFor position records resources rows)),
  axioms: [propext]
}
