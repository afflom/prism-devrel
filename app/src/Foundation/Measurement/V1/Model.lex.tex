definition{
  name: excessAbove,
  parameters: [bound : Nat, value : Nat],
  result: Nat,
  body: (prim subtract value bound : Nat)
}

definition{
  name: excessBelow,
  parameters: [bound : Nat, value : Nat],
  result: Nat,
  body: (prim subtract bound value : Nat)
}

theorem{
  name: noExcessAboveIsAtMost,
  parameters: [bound : Nat, value : Nat],
  statement: (((excessAbove bound value) = 0) <-> (value <= bound)),
  proof: linear_arithmetic{definitions: [excessAbove]},
  axioms: [Classical.choice, Quot.sound, propext]
}

theorem{
  name: noExcessBelowIsAtLeast,
  parameters: [bound : Nat, value : Nat],
  statement: (((excessBelow bound value) = 0) <-> (bound <= value)),
  proof: linear_arithmetic{definitions: [excessBelow]},
  axioms: [Classical.choice, Quot.sound, propext]
}

theorem{
  name: excessAboveFallsWithTheValue,
  parameters: [bound : Nat, before : Nat, after : Nat],
  statement: ((after <= before) -> ((excessAbove bound after) <= (excessAbove bound before))),
  proof: linear_arithmetic{definitions: [excessAbove]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: excessBelowFallsAsTheValueRises,
  parameters: [bound : Nat, before : Nat, after : Nat],
  statement: ((before <= after) -> ((excessBelow bound after) <= (excessBelow bound before))),
  proof: linear_arithmetic{definitions: [excessBelow]},
  axioms: [Quot.sound, propext]
}

definition{
  name: excessOver,
  parameters: [criteria : DecisionCriteria, value : Nat],
  result: Nat,
  body: (match (project criteria direction)
    | Direction.atMost => (excessAbove (project criteria bound) value)
    | Direction.atLeast => (excessBelow (project criteria bound) value))
}

definition{
  name: satisfies,
  parameters: [criteria : DecisionCriteria, value : Nat],
  result: Bool,
  body: ((excessOver criteria value) =? 0)
}

definition{
  name: indicatorsKnown,
  parameters: [
    conceptCount : Nat,
    derivedCount : Nat,
    criteriaCount : Nat,
    indicators : (List Indicator)
  ],
  result: Bool,
  body: (match indicators
    | List.nil => true
    | List.cons item rest => (((project item conceptIndex) <? conceptCount) && (((project item derivedMeasureIndex) <? derivedCount) && (((project item criteriaIndex) <? criteriaCount) && (indicatorsKnown conceptCount derivedCount criteriaCount rest))))),
  recursive_argument: indicators
}

definition{
  name: measured,
  parameters: [indicator : Nat, results : (List MeasurementResult)],
  result: Bool,
  body: (match results
    | List.nil => false
    | List.cons item rest => (((project item indicatorIndex) =? indicator) || (measured indicator rest))),
  recursive_argument: results
}

definition{
  name: everyIndicatorMeasured,
  parameters: [indicators : (List Indicator), results : (List MeasurementResult)],
  result: Bool,
  body: (match indicators
    | List.nil => true
    | List.cons item rest => ((measured (project item index) results) && (everyIndicatorMeasured rest results))),
  recursive_argument: indicators
}
